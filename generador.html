    // Detecta el municipio más cercano a unas coordenadas
    function detectarMunicipioPorCoords(lat, lng) {
        if(!estadoActivo) return null;
        var mejor = null, minDist = Infinity;
        // Solo municipios principales (no subparadas): IDs < idBase*100 y dentro del rango
        Object.keys(coordenadasLocalidades).forEach(function(k){
            var idn = parseInt(k);
            if(idn >= estadoActivo.idBase && idn <= estadoActivo.idMax){
                // Es subparada si: idn >= estadoActivo.idBase*100 || idn % 100 !== 0 && idn > 99
                // Simplificado: si idn tiene más de 2 dígitos y no coincide con idBase
                var esMunicipioPrincipal = (idn < 100) || (idn === estadoActivo.idBase) || 
                    (Object.keys(localidadesObj).indexOf(String(idn)) !== -1 && idn < estadoActivo.idBase * 100);
                if(esMunicipioPrincipal){
                    var c = coordenadasLocalidades[idn];
                    if(c && c.lat !== 0){
                        var d = haversine(lat, lng, c.lat, c.lng);
                        if(d < minDist){ minDist = d; mejor = {id: idn, nombre: localidadesObj[idn], dist: d}; }
                    }
                }
            }
        });
        return mejor;
    }

    // Calcula el siguiente ID disponible para un municipio (base×100 + correlativo)
    function calcularSiguienteId(municipioId) {
        var base = municipioId * 100;
        var usados = [];
        // Revisar subparadas actuales
        subparadas.forEach(function(s){ if(s.id >= base && s.id < base + 100) usados.push(s.id); });
        // Revisar localidadesObj
        Object.keys(localidadesObj).map(Number).forEach(function(id){
            if(id >= base && id < base + 100) usados.push(id);
        });
        return usados.length > 0 ? Math.max.apply(null, usados) + 1 : base + 1;
    }

    // Agrega la subparada con el municipio ya confirmado
    function agregarConMunicipio(munId, munNombre, nombre, lat, lng){
        var idNuevo = calcularSiguienteId(munId);
        subparadas.push({ id: idNuevo, nombre: nombre, lat: lat, lng: lng, municipioId: munId, municipioNombre: munNombre });
        document.getElementById('inputCoordenadas').value = '';
        document.getElementById('inputNombre').value = '';
        document.getElementById('inputCoordenadas').focus();
        actualizarListaSubparadas();
        mostrarToast('✅ ID '+idNuevo+': '+nombre+' → '+munNombre+' (mun. '+munId+')');
    }

    // Muestra lista de municipios para selección manual
    function mostrarSelectorMunicipios(nombre, lat, lng){
        var idsEstado = Object.keys(localidadesObj).map(Number)
            .filter(function(id){ return id >= estadoActivo.idBase && id <= estadoActivo.idMax; })
            .sort(function(a,b){ return a-b; });
        if(idsEstado.length === 0) idsEstado = [estadoActivo.idBase];

        // Calcular distancias para ordenar
        var conDistancias = idsEstado.map(function(id){
            var c = coordenadasLocalidades[id];
            var d = c && c.lat ? haversine(lat, lng, c.lat, c.lng) : 999;
            return { id: id, nombre: localidadesObj[id] || 'ID '+id, dist: d };
        });
        conDistancias.sort(function(a,b){ return a.dist - b.dist; });

        var modal = document.createElement('div');
        modal.className = 'modal-overlay active';
        modal.id = 'modalSelectorMunicipio';
        var html = '
🏙️ Selecciona el municipio
';
        html += '
📍 '+nombre+'
';
        html += '
';
        conDistancias.slice(0, 20).forEach(function(mun){
            html += '';
        });
        html += '
';
        modal.innerHTML = html;
        document.body.appendChild(modal);
    }

    // Flujo principal: GPS
    function capturarGPS(){
        if(!navigator.geolocation){mostrarToast('⚠️ GPS no disponible',true);return;}
        mostrarToast('📡 Obteniendo GPS...');
        navigator.geolocation.getCurrentPosition(
            function(pos){
                var lat=pos.coords.latitude, lng=pos.coords.longitude;
                document.getElementById('inputCoordenadas').value=lat.toFixed(6)+', '+lng.toFixed(6);
                var nombre = prompt('✏️ Nombre de la localidad:','Localidad GPS');
                if(!nombre){ mostrarToast('⚠️ Cancelado',true); return; }
                document.getElementById('inputNombre').value = nombre;
                // Detectar municipio
                var mun = detectarMunicipioPorCoords(lat, lng);
                if(mun){
                    var conf = confirm('📍 ¿Es '+mun.nombre+'? (ID '+mun.id+')\nDistancia: '+mun.dist.toFixed(1)+' km');
                    if(conf){
                        agregarConMunicipio(mun.id, mun.nombre, nombre, lat, lng);
                        return;
                    }
                }
                // Si no detectó o dijo NO → selector manual
                mostrarSelectorMunicipios(nombre, lat, lng);
            },
            function(err){mostrarToast('⚠️ Error GPS',true);},
            {enableHighAccuracy:true,timeout:15000,maximumAge:0}
        );
    }

    // Flujo principal: Manual
    function agregarSubparada(){
        if(!estadoActivo) return;
        var coord = document.getElementById('inputCoordenadas').value.trim();
        var nombre = document.getElementById('inputNombre').value.trim();
        if(!coord){ mostrarToast('⚠️ Pega coordenadas o usa GPS', true); return; }
        if(!nombre){ mostrarToast('⚠️ Pon el nombre', true); return; }
        var partes = coord.split(/[,;\s]+/).filter(function(p){ return p.length>0; });
        if(partes.length<2){ mostrarToast('⚠️ Formato: lat, lng', true); return; }
        var lat = parseFloat(partes[0]), lng = parseFloat(partes[1]);
        if(isNaN(lat)||isNaN(lng)){ mostrarToast('⚠️ Coordenadas inválidas', true); return; }

        // Detectar municipio
        var mun = detectarMunicipioPorCoords(lat, lng);
        if(mun){
            var conf = confirm('📍 ¿Es '+mun.nombre+'? (ID '+mun.id+')\nDistancia: '+mun.dist.toFixed(1)+' km');
            if(conf){
                agregarConMunicipio(mun.id, mun.nombre, nombre, lat, lng);
                return;
            }
        }
        // Si no detectó o dijo NO → selector manual
        mostrarSelectorMunicipios(nombre, lat, lng);
    }

    function generarTXT(tipo){
        tipoGeneracion=tipo;cerrarModal();
        if(!municipioActivo) municipioActivo={id:estadoActivo.idBase,nombre:localidadesObj[estadoActivo.idBase]||estadoActivo.nombre};
        var txt='',bKmVar='bKm_'+estadoActivo.clave,bKmValor=estadoActivo.bKm;

        // Agrupar por municipio
        var porMunicipio = {};
        subparadas.forEach(function(s){
            var munId = s.municipioId || municipioActivo.id;
            if(!porMunicipio[munId]) porMunicipio[munId] = [];
            porMunicipio[munId].push(s);
        });

        if(tipo==='completa'){
            txt+='// ============================================================\n// MATRIZ DE TARIFAS — MÉXICO\n// ============================================================\n\n';
            var todos=estadosDetectados.slice();if(estadoActivo.esNuevo)todos.push(estadoActivo);
            txt+='// ===== TARIFAS POR ESTADO =====\n';todos.forEach(function(e){txt+='const bKm_'+e.clave+' = '+e.bKm+';\n';});
            txt+='\nconst localidadesObj = {\n';Object.keys(localidadesObj).forEach(function(id){txt+='    '+id+': "'+localidadesObj[id]+'",\n';});
            subparadas.forEach(function(s){txt+='    '+s.id+': "'+s.nombre+'",\n';});
            txt+='};\n\nconst coordenadasLocalidades = {\n';Object.keys(coordenadasLocalidades).forEach(function(id){var c=coordenadasLocalidades[id];txt+='    '+id+': { lat: '+c.lat+', lng: '+c.lng+' },\n';});
            subparadas.forEach(function(s){txt+='    '+s.id+': { lat: '+s.lat+', lng: '+s.lng+' },\n';});
            txt+='};\n\nconst MATRIZ_QUERETARO = {\n';
            Object.keys(MATRIZ_QUERETARO).forEach(function(k){var origen=parseInt(k.split('-')[0]);txt+='    "'+k+'":{km:'+MATRIZ_QUERETARO[k].km+', tarifa: '+MATRIZ_QUERETARO[k].km+' * '+obtenerClaveEstadoPorId(origen)+'},\n';});
            Object.keys(porMunicipio).forEach(function(munId){
                var nombreMun = localidadesObj[munId] || 'Municipio '+munId;
                txt+='\n    // --- '+nombreMun+' ---\n';
                porMunicipio[munId].forEach(function(s){txt+='    "'+munId+'-'+s.id+'":{km:0, tarifa: 0 * '+bKmVar+'},  // '+s.nombre+'\n';});
            });
            txt+='};\n';
        } else {
            txt+='// ===== '+estadoActivo.nombre.toUpperCase()+' =====\nconst '+bKmVar+' = '+bKmValor+';\n\n';
            txt+='// --- Agregar en localidadesObj ---\n';
            subparadas.forEach(function(s){txt+=s.id+': "'+s.nombre+'",\n';});
            txt+='\n// --- Agregar en coordenadasLocalidades ---\n';
            subparadas.forEach(function(s){txt+=s.id+': { lat: '+s.lat+', lng: '+s.lng+' },\n';});
            txt+='\n// --- Agregar en MATRIZ_QUERETARO ---\n';
            Object.keys(porMunicipio).forEach(function(munId){
                var nombreMun = localidadesObj[munId] || 'Municipio '+munId;
                txt+='\n    // --- '+nombreMun+' ---\n';
                porMunicipio[munId].forEach(function(s){txt+='"'+munId+'-'+s.id+'":{km:0, tarifa: 0 * '+bKmVar+'},  // '+s.nombre+'\n';});
